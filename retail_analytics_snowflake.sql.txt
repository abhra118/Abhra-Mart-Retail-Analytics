CREATE OR REPLACE TABLE ANALYTICS.DIM_DATE AS

WITH DATE_BOUNDS AS (
    SELECT
        MIN(ORDER_DATE::DATE) AS START_DATE,
        MAX(ORDER_DATE::DATE) AS END_DATE
    FROM NOVAMART_DB.STAGING.STG_ORDERS
),

DATE_SERIES AS (
    SELECT
        DATEADD(
            DAY,
            SEQ4(),
            START_DATE
        )::DATE AS FULL_DATE,
        END_DATE
    FROM DATE_BOUNDS,
         TABLE(GENERATOR(ROWCOUNT => 1000))
)

SELECT
    TO_NUMBER(
        TO_CHAR(FULL_DATE, 'YYYYMMDD')
    ) AS DATE_KEY,

    FULL_DATE,
    DAY(FULL_DATE) AS DAY_NUMBER,
    DAYNAME(FULL_DATE) AS DAY_NAME,
    WEEKOFYEAR(FULL_DATE) AS WEEK_NUMBER,
    MONTH(FULL_DATE) AS MONTH_NUMBER,
    MONTHNAME(FULL_DATE) AS MONTH_NAME,
    QUARTER(FULL_DATE) AS QUARTER_NUMBER,
    YEAR(FULL_DATE) AS YEAR_NUMBER,

    IFF(
        DAYOFWEEKISO(FULL_DATE) IN (6, 7),
        TRUE,
        FALSE
    ) AS IS_WEEKEND

FROM DATE_SERIES
WHERE FULL_DATE <= END_DATE;
CREATE OR REPLACE TABLE ANALYTICS.FACT_SALES AS

SELECT
    od.ORDER_DETAIL_ID,
    o.ORDER_ID,

    COALESCE(dd.DATE_KEY, -1) AS ORDER_DATE_KEY,
    COALESCE(dc.CUSTOMER_KEY, -1) AS CUSTOMER_KEY,
    COALESCE(ds.STORE_KEY, -1) AS STORE_KEY,
    COALESCE(dp.PRODUCT_KEY, -1) AS PRODUCT_KEY,

    o.ORDER_DATE,
    o.ORDER_STATUS,
    o.PAYMENT_METHOD,

    od.QUANTITY,
    od.UNIT_PRICE,
    od.DISCOUNT,
    od.GROSS_AMOUNT,
    od.DISCOUNT_AMOUNT,
    od.NET_AMOUNT,

    od.PRICE_WAS_CORRECTED,
    o.CUSTOMER_ID_WAS_MISSING,
    o.STORE_ID_WAS_MISSING

FROM NOVAMART_DB.STAGING.STG_ORDER_DETAILS od

INNER JOIN NOVAMART_DB.STAGING.STG_ORDERS o
    ON od.ORDER_ID = o.ORDER_ID

LEFT JOIN ANALYTICS.DIM_DATE dd
    ON o.ORDER_DATE::DATE = dd.FULL_DATE

LEFT JOIN ANALYTICS.DIM_CUSTOMER dc
    ON o.CUSTOMER_ID = dc.CUSTOMER_ID

LEFT JOIN ANALYTICS.DIM_STORE ds
    ON o.STORE_ID = ds.STORE_ID

LEFT JOIN ANALYTICS.DIM_PRODUCT dp
    ON od.PRODUCT_ID = dp.PRODUCT_ID;
    SELECT
    COUNT(*) AS FACT_ROWS,
    COUNT(DISTINCT ORDER_DETAIL_ID) AS DISTINCT_ORDER_DETAILS,
    COUNT(DISTINCT ORDER_ID) AS ORDERS_WITH_VALID_LINES,
    SUM(NET_AMOUNT) AS TOTAL_NET_AMOUNT
FROM ANALYTICS.FACT_SALES;
SELECT
    COUNT_IF(CUSTOMER_KEY = -1) AS UNKNOWN_CUSTOMER_ROWS,
    COUNT_IF(STORE_KEY = -1) AS UNKNOWN_STORE_ROWS,
    COUNT_IF(PRODUCT_KEY = -1) AS UNKNOWN_PRODUCT_ROWS,
    COUNT_IF(ORDER_DATE_KEY = -1) AS UNKNOWN_DATE_ROWS
FROM ANALYTICS.FACT_SALES;
CREATE OR REPLACE TABLE ANALYTICS.FACT_PAYMENTS AS

SELECT
    p.PAYMENT_ID,
    p.ORDER_ID,

    COALESCE(dd.DATE_KEY, -1) AS PAYMENT_DATE_KEY,
    COALESCE(dc.CUSTOMER_KEY, -1) AS CUSTOMER_KEY,
    COALESCE(ds.STORE_KEY, -1) AS STORE_KEY,

    p.PAYMENT_DATE,
    p.PAYMENT_STATUS,
    p.AMOUNT,
    p.AMOUNT_WAS_NEGATIVE

FROM NOVAMART_DB.STAGING.STG_PAYMENTS p

INNER JOIN NOVAMART_DB.STAGING.STG_ORDERS o
    ON p.ORDER_ID = o.ORDER_ID

LEFT JOIN ANALYTICS.DIM_DATE dd
    ON p.PAYMENT_DATE::DATE = dd.FULL_DATE

LEFT JOIN ANALYTICS.DIM_CUSTOMER dc
    ON o.CUSTOMER_ID = dc.CUSTOMER_ID

LEFT JOIN ANALYTICS.DIM_STORE ds
    ON o.STORE_ID = ds.STORE_ID;
    SELECT
    COUNT(*) AS FACT_ROWS,
    COUNT(DISTINCT PAYMENT_ID) AS DISTINCT_PAYMENTS,
    COUNT(DISTINCT ORDER_ID) AS ORDERS_WITH_PAYMENTS,
    SUM(AMOUNT) AS TOTAL_PAYMENT_AMOUNT
FROM ANALYTICS.FACT_PAYMENTS;
SELECT
    COUNT_IF(PAYMENT_DATE_KEY = -1) AS UNKNOWN_DATES,
    COUNT_IF(CUSTOMER_KEY = -1) AS UNKNOWN_CUSTOMERS,
    COUNT_IF(STORE_KEY = -1) AS UNKNOWN_STORES
FROM ANALYTICS.FACT_PAYMENTS;
CREATE OR REPLACE TABLE ANALYTICS.FACT_RETURNS AS

SELECT
    r.RETURN_ID,
    r.ORDER_ID,

    COALESCE(dd.DATE_KEY, -1) AS RETURN_DATE_KEY,
    COALESCE(dc.CUSTOMER_KEY, -1) AS CUSTOMER_KEY,
    COALESCE(ds.STORE_KEY, -1) AS STORE_KEY,

    r.RETURN_DATE,
    r.RETURN_REASON,
    r.REFUND_STATUS

FROM NOVAMART_DB.STAGING.STG_RETURNS r

INNER JOIN NOVAMART_DB.STAGING.STG_ORDERS o
    ON r.ORDER_ID = o.ORDER_ID

LEFT JOIN NOVAMART_DB.ANALYTICS.DIM_DATE dd
    ON r.RETURN_DATE = dd.FULL_DATE

LEFT JOIN NOVAMART_DB.ANALYTICS.DIM_CUSTOMER dc
    ON o.CUSTOMER_ID = dc.CUSTOMER_ID

LEFT JOIN NOVAMART_DB.ANALYTICS.DIM_STORE ds
    ON o.STORE_ID = ds.STORE_ID;
    SELECT
    COUNT(*) AS FACT_ROWS,
    COUNT(DISTINCT RETURN_ID) AS DISTINCT_RETURNS,
    COUNT(DISTINCT ORDER_ID) AS RETURNED_ORDERS
FROM ANALYTICS.FACT_RETURNS;
SELECT
    COUNT_IF(RETURN_DATE_KEY = -1) AS UNKNOWN_DATES,
    COUNT_IF(CUSTOMER_KEY = -1) AS UNKNOWN_CUSTOMERS,
    COUNT_IF(STORE_KEY = -1) AS UNKNOWN_STORES
FROM ANALYTICS.FACT_RETURNS;
CREATE OR REPLACE TABLE ANALYTICS.FACT_INVENTORY AS

SELECT
    COALESCE(dd.DATE_KEY, -1) AS SNAPSHOT_DATE_KEY,
    COALESCE(ds.STORE_KEY, -1) AS STORE_KEY,
    COALESCE(dp.PRODUCT_KEY, -1) AS PRODUCT_KEY,

    i.SNAPSHOT_DATE,
    i.STOCK_ON_HAND,
    i.REORDER_LEVEL,

    IFF(
        i.STOCK_ON_HAND <= i.REORDER_LEVEL,
        TRUE,
        FALSE
    ) AS REORDER_REQUIRED,

    GREATEST(
        i.REORDER_LEVEL - i.STOCK_ON_HAND,
        0
    ) AS REORDER_SHORTAGE_QUANTITY

FROM NOVAMART_DB.STAGING.STG_INVENTORY i

LEFT JOIN ANALYTICS.DIM_DATE dd
    ON i.SNAPSHOT_DATE = dd.FULL_DATE

LEFT JOIN ANALYTICS.DIM_STORE ds
    ON i.STORE_ID = ds.STORE_ID

LEFT JOIN ANALYTICS.DIM_PRODUCT dp
    ON i.PRODUCT_ID = dp.PRODUCT_ID;
    SELECT
    COUNT(*) AS FACT_ROWS,
    COUNT_IF(REORDER_REQUIRED) AS PRODUCTS_REQUIRING_REORDER,
    SUM(STOCK_ON_HAND) AS TOTAL_STOCK_ON_HAND,
    SUM(REORDER_SHORTAGE_QUANTITY) AS TOTAL_SHORTAGE_QUANTITY
FROM ANALYTICS.FACT_INVENTORY;
SELECT
    COUNT_IF(SNAPSHOT_DATE_KEY = -1) AS UNKNOWN_DATES,
    COUNT_IF(STORE_KEY = -1) AS UNKNOWN_STORES,
    COUNT_IF(PRODUCT_KEY = -1) AS UNKNOWN_PRODUCTS
FROM ANALYTICS.FACT_INVENTORY;
SELECT 'DIM_CUSTOMER' AS TABLE_NAME, COUNT(*) AS ROW_COUNT
FROM ANALYTICS.DIM_CUSTOMER

UNION ALL
SELECT 'DIM_STORE', COUNT(*)
FROM ANALYTICS.DIM_STORE

UNION ALL

FROM ANALYTICS.DIM_PRODUCT

UNION ALL
SELECT 'DIM_DATE', COUNT(*)

UNION ALL

FROM ANALYTICS.FACT_SALES

FROM ANALYTICS.FACT_PAYMENTS

UNION ALL

UNION ALL

FROM ANALYTICS.FACT_INVENTORY

CREATE SCHEMA IF NOT EXISTS NOVAMART_DB.MART;

CREATE OR REPLACE VIEW MART.DAILY_SALES_SUMMARY AS

SELECT
    d.DATE_KEY,
    d.FULL_DATE,
    d.DAY_NAME,
    d.WEEK_NUMBER,
    d.MONTH_NUMBER,
    d.MONTH_NAME,
    d.QUARTER_NUMBER,
    d.YEAR_NUMBER,
    d.IS_WEEKEND,

    COUNT(DISTINCT f.ORDER_ID) AS TOTAL_ORDERS,

    COUNT(DISTINCT IFF(
        f.ORDER_STATUS = 'Completed',
        f.ORDER_ID,
        NULL
    )) AS COMPLETED_ORDERS,

    COUNT(DISTINCT IFF(
        f.ORDER_STATUS = 'Cancelled',
        f.ORDER_ID,
        NULL
    )) AS CANCELLED_ORDERS,

    COUNT(DISTINCT IFF(
        f.ORDER_STATUS = 'Pending',
        f.ORDER_ID,
        NULL
    )) AS PENDING_ORDERS,

    COUNT(DISTINCT IFF(
        f.ORDER_STATUS = 'Returned',
        f.ORDER_ID,
        NULL
    )) AS RETURNED_ORDERS,

    SUM(f.GROSS_AMOUNT) AS GROSS_SALES,
    SUM(f.DISCOUNT_AMOUNT) AS DISCOUNT_AMOUNT,
    SUM(f.NET_AMOUNT) AS NET_SALES,

    SUM(IFF(
        f.ORDER_STATUS = 'Completed',
        f.NET_AMOUNT,
        0
    )) AS COMPLETED_SALES,

    DIV0(
        SUM(IFF(
            f.ORDER_STATUS = 'Completed',
            f.NET_AMOUNT,
            0
        )),
        COUNT(DISTINCT IFF(
            f.ORDER_STATUS = 'Completed',
            f.ORDER_ID,
            NULL
        ))
    ) AS AVERAGE_COMPLETED_ORDER_VALUE

FROM ANALYTICS.FACT_SALES f

INNER JOIN ANALYTICS.DIM_DATE d
    ON f.ORDER_DATE_KEY = d.DATE_KEY

GROUP BY
    d.DATE_KEY,
    d.FULL_DATE,
    d.DAY_NAME,
    d.WEEK_NUMBER,
    d.MONTH_NUMBER,
    d.MONTH_NAME,
    d.QUARTER_NUMBER,
    d.YEAR_NUMBER,
    d.IS_WEEKEND;
    SELECT
    SUM(TOTAL_ORDERS) AS TOTAL_ORDERS,
    SUM(COMPLETED_ORDERS) AS COMPLETED_ORDERS,
    SUM(CANCELLED_ORDERS) AS CANCELLED_ORDERS,
    SUM(PENDING_ORDERS) AS PENDING_ORDERS,
    SUM(RETURNED_ORDERS) AS RETURNED_ORDERS,
    SUM(COMPLETED_SALES) AS COMPLETED_SALES
FROM MART.DAILY_SALES_SUMMARY;
CREATE OR REPLACE VIEW MART.STORE_PERFORMANCE AS

WITH SALES_AGG AS (
    SELECT
        STORE_KEY,
        COUNT(DISTINCT ORDER_ID) AS TOTAL_ORDERS,
        COUNT(DISTINCT CUSTOMER_KEY) AS DISTINCT_CUSTOMERS,
        SUM(QUANTITY) AS UNITS_SOLD,
        SUM(GROSS_AMOUNT) AS GROSS_SALES,
        SUM(DISCOUNT_AMOUNT) AS DISCOUNT_AMOUNT,
        SUM(NET_AMOUNT) AS NET_SALES,

        COUNT(DISTINCT IFF(
            ORDER_STATUS = 'Completed',
            ORDER_ID,
            NULL
        )) AS COMPLETED_ORDERS,

        SUM(IFF(
            ORDER_STATUS = 'Completed',
            NET_AMOUNT,
            0
        )) AS COMPLETED_SALES

    FROM ANALYTICS.FACT_SALES
    GROUP BY STORE_KEY
),

RETURN_AGG AS (
    SELECT
        STORE_KEY,
        COUNT(*) AS TOTAL_RETURNS,
        COUNT(DISTINCT ORDER_ID) AS RETURNED_ORDERS,
        COUNT_IF(REFUND_STATUS = 'Processed') AS PROCESSED_REFUNDS
    FROM ANALYTICS.FACT_RETURNS
    GROUP BY STORE_KEY
),

INVENTORY_AGG AS (
    SELECT
        STORE_KEY,
        SUM(STOCK_ON_HAND) AS STOCK_ON_HAND,
        COUNT_IF(REORDER_REQUIRED) AS PRODUCTS_REQUIRING_REORDER,
        SUM(REORDER_SHORTAGE_QUANTITY) AS SHORTAGE_QUANTITY
    FROM ANALYTICS.FACT_INVENTORY
    GROUP BY STORE_KEY
)

SELECT
    s.STORE_KEY,
    s.STORE_ID,
    s.STORE_NAME,
    s.CITY,
    s.STATE,
    s.STORE_TYPE,

    COALESCE(sa.TOTAL_ORDERS, 0) AS TOTAL_ORDERS,
    COALESCE(sa.COMPLETED_ORDERS, 0) AS COMPLETED_ORDERS,
    COALESCE(sa.DISTINCT_CUSTOMERS, 0) AS DISTINCT_CUSTOMERS,
    COALESCE(sa.UNITS_SOLD, 0) AS UNITS_SOLD,
    COALESCE(sa.GROSS_SALES, 0) AS GROSS_SALES,
    COALESCE(sa.DISCOUNT_AMOUNT, 0) AS DISCOUNT_AMOUNT,
    COALESCE(sa.NET_SALES, 0) AS NET_SALES,
    COALESCE(sa.COMPLETED_SALES, 0) AS COMPLETED_SALES,

    DIV0(
        COALESCE(sa.COMPLETED_SALES, 0),
        COALESCE(sa.COMPLETED_ORDERS, 0)
    ) AS AVERAGE_COMPLETED_ORDER_VALUE,

    COALESCE(ra.TOTAL_RETURNS, 0) AS TOTAL_RETURNS,
    COALESCE(ra.RETURNED_ORDERS, 0) AS RETURNED_ORDERS,
    COALESCE(ra.PROCESSED_REFUNDS, 0) AS PROCESSED_REFUNDS,

    DIV0(
        COALESCE(ra.RETURNED_ORDERS, 0),
        COALESCE(sa.TOTAL_ORDERS, 0)
    ) AS RETURN_RATE,

    COALESCE(ia.STOCK_ON_HAND, 0) AS STOCK_ON_HAND,
    COALESCE(ia.PRODUCTS_REQUIRING_REORDER, 0)
        AS PRODUCTS_REQUIRING_REORDER,
    COALESCE(ia.SHORTAGE_QUANTITY, 0) AS SHORTAGE_QUANTITY

FROM ANALYTICS.DIM_STORE s

LEFT JOIN SALES_AGG sa
    ON s.STORE_KEY = sa.STORE_KEY

LEFT JOIN RETURN_AGG ra
    ON s.STORE_KEY = ra.STORE_KEY

LEFT JOIN INVENTORY_AGG ia
    ON s.STORE_KEY = ia.STORE_KEY;
    SELECT
    STORE_NAME,
    CITY,
    COMPLETED_ORDERS,
    COMPLETED_SALES,
    RETURN_RATE,
    PRODUCTS_REQUIRING_REORDER
FROM MART.STORE_PERFORMANCE
WHERE STORE_KEY <> -1
ORDER BY COMPLETED_SALES DESC
LIMIT 10;
CREATE OR REPLACE VIEW MART.PRODUCT_PERFORMANCE AS

WITH SALES_AGG AS (
    SELECT
        PRODUCT_KEY,
        COUNT(DISTINCT ORDER_ID) AS TOTAL_ORDERS,
        COUNT(DISTINCT CUSTOMER_KEY) AS DISTINCT_CUSTOMERS,
        SUM(QUANTITY) AS UNITS_SOLD,
        SUM(GROSS_AMOUNT) AS GROSS_SALES,
        SUM(DISCOUNT_AMOUNT) AS DISCOUNT_AMOUNT,
        SUM(NET_AMOUNT) AS NET_SALES,

        SUM(IFF(
            ORDER_STATUS = 'Completed',
            QUANTITY,
            0
        )) AS COMPLETED_UNITS_SOLD,

        SUM(IFF(
            ORDER_STATUS = 'Completed',
            NET_AMOUNT,
            0
        )) AS COMPLETED_SALES

CREATE OR REPLACE VIEW MART.INVENTORY_REORDER_ALERTS AS

WITH LATEST_INVENTORY AS (
    SELECT *
    FROM ANALYTICS.FACT_INVENTORY

    QUALIFY ROW_NUMBER() OVER (
        PARTITION BY STORE_KEY, PRODUCT_KEY
        ORDER BY SNAPSHOT_DATE DESC
    ) = 1
),

PROJECT_PERIOD AS (
    SELECT
        DATEDIFF(
            DAY,
            MIN(FULL_DATE),
            MAX(FULL_DATE)
        ) + 1 AS TOTAL_DAYS
    FROM ANALYTICS.DIM_DATE
),

SALES_VELOCITY AS (
    SELECT
        STORE_KEY,
        PRODUCT_KEY,

        SUM(IFF(
            ORDER_STATUS = 'Completed',
            QUANTITY,
            0
        )) AS COMPLETED_UNITS_SOLD

    FROM ANALYTICS.FACT_SALES
    GROUP BY STORE_KEY, PRODUCT_KEY
)

SELECT
    i.SNAPSHOT_DATE,
    s.STORE_ID,
    s.STORE_NAME,
    s.CITY,
    s.STATE,
    s.STORE_TYPE,

    p.PRODUCT_ID,
    p.PRODUCT_NAME,
    p.CATEGORY,
    p.PRODUCT_STATUS,

    i.STOCK_ON_HAND,
    i.REORDER_LEVEL,
    i.REORDER_SHORTAGE_QUANTITY,

    CASE
        WHEN i.STOCK_ON_HAND = 0
            THEN 'Out of Stock'
        WHEN i.STOCK_ON_HAND <= i.REORDER_LEVEL * 0.5
            THEN 'Critical'
        ELSE 'Reorder Required'
    END AS INVENTORY_ALERT_LEVEL,

    COALESCE(v.COMPLETED_UNITS_SOLD, 0) AS COMPLETED_UNITS_SOLD,

    ROUND(
        DIV0(
            COALESCE(v.COMPLETED_UNITS_SOLD, 0),
            pp.TOTAL_DAYS
        ),
        2
    ) AS AVERAGE_DAILY_UNITS_SOLD,

    CASE
        WHEN COALESCE(v.COMPLETED_UNITS_SOLD, 0) = 0
            THEN NULL
        ELSE ROUND(
            i.STOCK_ON_HAND /
            (
                v.COMPLETED_UNITS_SOLD /
                pp.TOTAL_DAYS
            ),
            1
        )
    END AS ESTIMATED_DAYS_OF_STOCK

FROM LATEST_INVENTORY i

INNER JOIN ANALYTICS.DIM_STORE s
    ON i.STORE_KEY = s.STORE_KEY

INNER JOIN ANALYTICS.DIM_PRODUCT p
    ON i.PRODUCT_KEY = p.PRODUCT_KEY

LEFT JOIN SALES_VELOCITY v
    ON i.STORE_KEY = v.STORE_KEY
   AND i.PRODUCT_KEY = v.PRODUCT_KEY

CROSS JOIN PROJECT_PERIOD pp

WHERE i.REORDER_REQUIRED = TRUE;
SELECT *
FROM MART.INVENTORY_REORDER_ALERTS
ORDER BY
    CASE INVENTORY_ALERT_LEVEL
        WHEN 'Out of Stock' THEN 1
        WHEN 'Critical' THEN 2
        ELSE 3
    END,
    REORDER_SHORTAGE_QUANTITY DESC

WITH CHECKS AS (

    SELECT
        'Fact order row count' AS CHECK_NAME,
        100000::NUMBER AS EXPECTED_VALUE,
        COUNT(*)::NUMBER AS ACTUAL_VALUE
    FROM ANALYTICS.FACT_ORDERS

    UNION ALL

    SELECT
        'Distinct order IDs',
        100000,
        COUNT(DISTINCT ORDER_ID)
    FROM ANALYTICS.FACT_ORDERS

    UNION ALL

    SELECT
        'Daily mart order reconciliation',
        (SELECT COUNT(*) FROM ANALYTICS.FACT_ORDERS),
        (SELECT SUM(TOTAL_ORDERS)
         FROM MART.DAILY_SALES_SUMMARY)

    UNION ALL

    SELECT
        'Store mart order reconciliation',
        (SELECT COUNT(*) FROM ANALYTICS.FACT_ORDERS),
        (SELECT SUM(TOTAL_ORDERS)
         FROM MART.STORE_PERFORMANCE)

    UNION ALL

    SELECT
        'Customer mart order reconciliation',
        (SELECT COUNT(*) FROM ANALYTICS.FACT_ORDERS),
        (SELECT SUM(TOTAL_ORDERS)
         FROM MART.CUSTOMER_PERFORMANCE)

    UNION ALL

    SELECT
        'Sales fact row count',
        150133,
        COUNT(*)
    FROM ANALYTICS.FACT_SALES

    UNION ALL

    SELECT
        'Payment fact row count',
        100000,
        COUNT(*)
    FROM ANALYTICS.FACT_PAYMENTS

    UNION ALL

    SELECT
        'Return fact row count',
        4978,
        COUNT(*)
    FROM ANALYTICS.FACT_RETURNS

    UNION ALL

    SELECT
        'Inventory fact row count',
        50000,
        COUNT(*)
    FROM ANALYTICS.FACT_INVENTORY
)

SELECT
    CHECK_NAME,
    EXPECTED_VALUE,
    ACTUAL_VALUE,

    IFF(
        EXPECTED_VALUE = ACTUAL_VALUE,
        'PASS',
        'FAIL'
    ) AS CHECK_STATUS

FROM CHECKS
ORDER BY CHECK_NAME;
